\begin{frame}[t]{De los abiertos a los marcos}

\small

% ========================================
% OPERACIONES ENTRE ABIERTOS
% ========================================
\uncover<1->{
Sea $X$ un espacio topológico. En $\mathcal O(X)$
consideramos las operaciones
}

\begin{center}
\uncover<1->{
$\displaystyle
\bigvee_{i\in I}U_i=\bigcup_{i\in I}U_i,
\qquad
U\wedge V=U\cap V.
$
}
\end{center}

\vspace{-0.15cm}

% ========================================
% DISTRIBUTIVIDAD
% ========================================
\uncover<2->{
Estas operaciones satisfacen la identidad
}

\begin{center}
\uncover<2->{
$\displaystyle
U\cap\left(\bigcup_{i\in I}V_i\right)
=
\bigcup_{i\in I}(U\cap V_i).
$
}
\end{center}

\vspace{-0.15cm}

% ========================================
% DEFINICIÓN DE MARCO
% ========================================
\uncover<3->{
\begin{block}{Definición}
Un \textbf{marco} es una retícula completa $A$
en la que, para todo $a\in A$ y $S\subseteq A$,
\[
a\wedge\bigvee S
=
\bigvee_{s\in S}(a\wedge s).
\]
\end{block}
}

% ========================================
% CONCLUSIÓN
% ========================================
\uncover<4->{
\begin{center}
\textbf{En particular, $\mathcal O(X)$ es un marco.}
\end{center}
}

\end{frame}


\begin{frame}[t]{¿Podemos recuperar los puntos?}

\small

% ========================================
% DE ESPACIOS A MARCOS
% ========================================
\uncover<1->{
A cada espacio topológico $X$ le asociamos
su marco de abiertos:
\[
X \longmapsto \mathcal O(X).
\]
}

% ========================================
% PUNTOS DE UN MARCO
% ========================================
\uncover<2->{
\begin{block}{Puntos de un marco}
Sea $A$ un marco. Un \textbf{punto} de $A$ es un
elemento $p\neq 1$ tal que
\[
a\wedge b\leq p
\quad\Longrightarrow\quad
a\leq p\; \text{o}\; b\leq p.
\]
Denotamos por $\operatorname{pt}(A)$
al conjunto de todos sus puntos.
\end{block}
}

% ========================================
% RELACIÓN ENTRE AMBAS CONSTRUCCIONES
% ========================================
\uncover<3->{
\begin{center}
\begin{tikzpicture}[
    >=Stealth,
    node distance=3.4cm,
    every node/.style={font=\normalsize}
]

\node (X) at (0,0) {$X$};
\node (OX) at (5,0) {$\mathcal O(X)$};

\draw[->,thick,bend left=25]
    (X) to node[above] {$\mathcal O$} (OX);

\draw[->,thick,bend left=25]
    (OX) to node[below] {$\operatorname{pt}$}
    (X);

\end{tikzpicture}
\end{center}
}
\end{frame}

\begin{frame}[t]{De subespacios a cocientes de marcos}

\small

% ========================================
% ESPACIO Y SUBESPACIO
% ========================================
\uncover<1->{

Consideremos el espacio
$X=\{x,y,z\}$ y el subconjunto $Y=\{x,z\}$.

\begin{center}
\begin{tikzpicture}[scale=0.75]

% Espacio X
\filldraw[
    fill=blue!10,
    draw=blue!55,
    thick,
    rounded corners=12pt
] (0,0) rectangle (5,2.5);

% Abierto {y,z}
\filldraw[
    fill=teal!12,
    draw=teal!65,
    thick,
    rounded corners=10pt
] (1.7,0.35) rectangle (4.7,2.15);

% Abierto {z}
\filldraw[
    fill=orange!15,
    draw=orange!80,
    thick,
    rounded corners=10pt
] (3.35,0.65) rectangle (4.45,1.85);

% Puntos
\fill (0.8,1.25) circle (2.5pt);
\fill (2.5,1.25) circle (2.5pt);
\fill (3.9,1.25) circle (2.5pt);

\node[below=4pt] at (0.8,1.25) {$x$};
\node[below=4pt] at (2.5,1.25) {$y$};
\node[below=4pt] at (3.9,1.25) {$z$};

% Subespacio Y: destacar sus puntos
\draw[
    red!75,
    thick,
    dashed
] (0.8,1.25) circle (0.27);

\draw[
    red!75,
    thick,
    dashed
] (3.9,1.25) circle (0.27);

\node[red!75!black] at (2.5,-0.35)
    {$Y=\{x,z\}$};

\end{tikzpicture}
\end{center}

}

% ========================================
% TOPOLOGÍA RELATIVA
% ========================================
\uncover<2->{

La topología relativa sobre $Y$ es
\[
\mathcal O(Y)
=
\{U\cap Y\mid U\in\mathcal O(X)\}
=
\{\varnothing,\{z\},Y\}.
\]
}
\vspace{-0.21cm}
% ========================================
% MORFISMO DE MARCOS
% ========================================
\uncover<3->{
La inclusión $Y\hookrightarrow X$ induce
un morfismo sobreyectivo de marcos:
\[
\begin{aligned}
\varphi\colon \mathcal O(X)&\longrightarrow\mathcal O(Y),\\
U&\longmapsto U\cap Y.
\end{aligned}
\]
}

% ========================================
% CONCLUSIÓN
% ========================================
\uncover<4->{
\begin{center}
\textbf{Los subespacios producen cocientes de marcos.}
\end{center}
}

\end{frame}


\begin{frame}[t]{De cocientes de marcos a núcleos}

\small

% ========================================
% RECORDATORIO
% ========================================
\uncover<1->{
Para $Y=\{x,z\}\subseteq X$, la aplicación
\(
\varphi\colon \mathcal O(X)\longrightarrow\mathcal O(Y),
\)
es un cociente de marcos.
}

% ========================================
% NÚCLEOS
% ========================================
\uncover<2->{
\begin{block}{Núcleos}
Un \textbf{núcleo} sobre un marco $A$ es una
aplicación $j:A\to A$ monótona, inflacionaria,
idempotente y que preserva ínfimos finitos.
\end{block}
}

% ========================================
% EJEMPLO
% ========================================
\uncover<3->{
El núcleo asociado al cociente anterior es
$j=\varphi_*\varphi$, y en nuestro ejemplo:

\begin{center}
\renewcommand{\arraystretch}{1.25}
\begin{tabular}{c|cccc}
$U$ &
$\varnothing$ & $\{z\}$ & $\{y,z\}$ & $X$\\
\hline
$j(U)$ &
$\varnothing$ & $\{y,z\}$ & $\{y,z\}$ & $X$
\end{tabular}
\end{center}
}

% ========================================
% CONCLUSIÓN
% ========================================
\uncover<4->{
\[
\mathcal{O}(X)_j
=\{\varnothing,\{y,z\},X\}
\cong\mathcal O(Y).
\]

\begin{center}
\textbf{Los núcleos generalizan la noción
de subespacio.}
\end{center}
}

\end{frame}


\begin{frame}[t]{De espacios a hiperespacios}

\small

% ========================================
% HIPERESPACIO DE CERRADOS
% ========================================
\uncover<1->{

En lugar de estudiar los puntos de $X$,
consideremos sus subconjuntos cerrados no vacíos:
\[
\mathcal F(X)=
\big\{
\{x\},\{x,y\},\{x,z\},X
\big\}.
\]

Cada elemento de $\mathcal F(X)$ será
un punto de un nuevo espacio topológico.

}

% ========================================
% TOPOLOGÍA DE VIETORIS
% ========================================
\uncover<2->{

\begin{block}{Topología de Vietoris}
Para cada $U\in\mathcal O(X)$, consideramos
\[
\Diamond U
=
\{F\in\mathcal F(X)\mid F\cap U\neq\varnothing\},
\qquad
\Box U
=
\{F\in\mathcal F(X)\mid F\subseteq U\}.
\]

Estos conjuntos generan una topología
sobre $\mathcal F(X)$.
\end{block}

}

% ========================================
% EJEMPLO CONCRETO
% ========================================
\uncover<3->{

Por ejemplo, para $U=\{z\}$,
\[
\Diamond\{z\}=\big\{\{x,z\},X\big\},
\qquad
\Box\{z\}=\varnothing.
\]

}

\end{frame}

\begin{frame}[t]{Vietoris sin puntos}

\small

% ========================================
% MOTIVACIÓN
% ========================================
\uncover<1->{

En la construcción clásica de Vietoris,
la topología del hiperespacio se obtiene
a partir de los abiertos de $X$:
\[
\mathcal O(X)
\longrightarrow
\mathcal O(\mathcal F(X)).
\]

}

\vspace{-0.2cm}

% ========================================
% CONSTRUCCIÓN POINT-FREE
% ========================================
\uncover<2->{

\begin{block}{La construcción de Vietoris}

Dado un marco $A$, consideramos los símbolos
\[
\lozenge a,\qquad \square a,
\qquad a\in A,
\]

sujetos a relaciones algebraicas que
generalizan las de la topología clásica (ver \cite{JohnstoneVietoris}).
\end{block}
}

\vspace{-0.25cm}
% ========================================
% CONCLUSIÓN
% ========================================
\uncover<3->{
El marco resultante se denota por $VA$.
\[
A\longmapsto VA
\]

\begin{center}
\textbf{Podemos construir hiperespacios
sin partir de un espacio de puntos.}
\end{center}

}

\end{frame}


\begin{frame}[t]{Modificando una topología}

\small

\uncover<1->{
Partimos nuevamente del espacio
\[
X=\{x,y,z\},\qquad
\tau=\{\varnothing,\{z\},\{y,z\},X\}.
\]
}
\uncover<2->{
\begin{block}{Modificación de Skula}
La topología de Skula está generada por
los abiertos de $X$ y sus complementos.
\end{block}
}

\uncover<3->{
En nuestro ejemplo, los conjuntos
\[
\begin{aligned}
\{x\}&=X\setminus\{y,z\},\quad
\{y\}&=\{y,z\}\setminus\{z\},\quad
\{z\}&\in\tau
\end{aligned}
\]
son abiertos en la nueva topología.
}
\uncover<4->{
\[
\tau_{\mathrm{Sk}}=\mathcal P(X)
\]

\begin{center}
\textbf{La modificación de Skula
convierte nuestro espacio en uno discreto.}
\end{center}
}

\end{frame}


\begin{frame}[t]{Modificaciones en marcos}

\small

% ========================================
% MOTIVACIÓN
% ========================================
\uncover<1->{
En un espacio topológico podemos construir
nuevas topologías incorporando abiertos.
En un marco, podemos realizar construcciones
análogas utilizando \textbf{núcleos}.

}

% ========================================
% NÚCLEOS BÁSICOS
% ========================================
\uncover<2->{

\begin{block}{Dos familias de núcleos}
Para cada $a\in A$, consideramos
\[
u_a(x)=a\vee x,
\qquad
v_a(x)=a\succ x.
\]
\end{block}
}

% ========================================
% CONSTRUCCIÓN PATCH
% ========================================
\uncover<3->{
Estos núcleos generan cocientes que denominamos 
cerrados y abiertos, respectivamente.


El \textbf{marco} $NA$ es generado por
los núcleos $u_a$ y $v_a$, con $a\in A$.
\[
A\longmapsto NA
\]

\begin{center}
\textbf{Una modificación topológica
descrita mediante operaciones algebraicas.}
\end{center}
}

\end{frame}


\begin{frame}[t]{¿Separación sin puntos?}

\small

% ========================================
% MOTIVACIÓN
% ========================================
\uncover<1->{

En topología clásica, los axiomas de
separación describen cómo distinguir
puntos mediante conjuntos abiertos.

}

% ========================================
% AXIOMAS CLÁSICOS
% ========================================
\uncover<2->{

\begin{block}{Algunos axiomas clásicos}

\begin{description}

\item[$T_0$:]
Dos puntos distintos pueden distinguirse
mediante un abierto.

\medskip

\item[$T_1$:]
Cada uno de dos puntos distintos tiene
una vecindad que no contiene al otro.

\medskip

\item[$T_2$:]
Dos puntos distintos admiten
vecindades abiertas disjuntas.

\end{description}

\end{block}

}

% ========================================
% PREGUNTA CENTRAL
% ========================================
\uncover<3->{

\begin{center}
\textbf{¿Podemos expresar estas propiedades
utilizando únicamente la estructura
de un marco?}
\end{center}

\medskip

La topología sin puntos propone
formulaciones algebraicas de los
axiomas de separación.

}

\end{frame}


\begin{frame}[t]{Reformulando la regularidad}

\small

\uncover<1->{

Un espacio $X$ es \textbf{regular} si para
cada $x\in U$, con $U$ abierto, existe
un abierto $V$ tal que
\[
x\in V\subseteq\overline V\subseteq U.
\]

}

\vspace{-0.15cm}

\uncover<2->{

\begin{block}{Una relación entre abiertos}

Para $V,U\in\mathcal O(X)$, escribimos
$V\prec U$ si existe $W\in\mathcal O(X)$
tal que
\[
V\cap W=\varnothing,
\qquad
U\cup W=X.
\]

\end{block}

}

\vspace{-0.15cm}

\uncover<3->{

La regularidad puede expresarse
únicamente mediante abiertos:
\[
\boxed{
U=\bigcup\{V\in\mathcal O(X):V\prec U\}.
}
\]

\begin{center}
\textbf{¡Los puntos han desaparecido
de la formulación!}
\end{center}

}

\end{frame}


\begin{frame}[t]{Regularidad sin puntos}

\small

\uncover<1->{

La formulación anterior utiliza
únicamente uniones e intersecciones
de abiertos.
\[
\begin{aligned}
V\cap W=\varnothing
&\longrightarrow b\wedge c=0,\\
U\cup W=X
&\longrightarrow a\vee c=1.
\end{aligned}
\]

}

\vspace{-0.15cm}

\uncover<2->{

\begin{block}{Relación bastante por debajo}

En un marco $A$, definimos
\[
b\prec a
\iff
\exists c\in A:
\quad b\wedge c=0,\quad a\vee c=1.
\]

\end{block}

}

\vspace{-0.15cm}

\uncover<3->{

\begin{block}{Marco regular}

Un marco $A$ es \textbf{regular} si

\[
a=\bigvee\{b\in A:b\prec a\}
\qquad
\text{para todo }a\in A.
\]

\end{block}

}

\end{frame}


\begin{frame}[t]{¿Una única noción de Hausdorff?}

\small

\uncover<1->{

En topología clásica, un espacio es
\textbf{Hausdorff} si dos puntos distintos
admiten vecindades abiertas disjuntas.

}

\uncover<2->{

En topología sin puntos existen
distintas formulaciones algebraicas
inspiradas en esta propiedad:

\[
\boxed{
(H)\qquad\text{y}\qquad(sH)
}
\]

}

\uncover<3->{

Estas condiciones no son equivalentes
en marcos arbitrarios.

\medskip

\begin{center}
\textbf{Una propiedad clásica puede admitir
distintas generalizaciones libres de puntos.}
\end{center}

}

\end{frame}


\begin{frame}[t]{Dos perspectivas de una misma teoría}

\small

\uncover<1->{

Los funtores de abiertos y de puntos
relacionan ambas teorías:
\[
\mathcal O\colon \mathbf{Top}\longrightarrow
\mathbf{Frm}^{op},
\qquad
\operatorname{pt}\colon \mathbf{Frm}^{op}
\longrightarrow\mathbf{Top}.
\]

}

\uncover<2->{

Para cada espacio $X$ y cada marco $A$,
tenemos aplicaciones canónicas:
\[
\eta_X\colon X\longrightarrow
\operatorname{pt}(\mathcal O(X)),
\]
\[
\varepsilon_A\colon A\longrightarrow
\mathcal O(\operatorname{pt}(A)).
\]

}

\uncover<3->{

\begin{block}{¿Cuándo recuperamos el objeto original?}

\begin{itemize}
    \item Si $X$ es sobrio, $\eta_X$
    es un homeomorfismo.

    \medskip

    \item Si $A$ es espacial, $\varepsilon_A$
    es un isomorfismo.
\end{itemize}

\end{block}

}

\end{frame}


\begin{frame}[t]{Una nueva forma de mirar la topología}

\small

\uncover<1->{

Hemos visto que distintas construcciones
clásicas admiten formulaciones algebraicas:

\[
\begin{aligned}
\text{Subespacios}
&\longrightarrow \text{Núcleos},\\[2mm]
\text{Hiperespacios}
&\longrightarrow \text{Marcos tipo Vietoris},\\[2mm]
\text{Modificaciones de espacios}
&\longrightarrow \text{Modificaciones de marcos}.
\end{aligned}
\]

}

\uncover<2->{

También encontramos formulaciones
libres de puntos de los axiomas
de separación.

}

\uncover<3->{

\begin{block}{Una idea para llevarse}

La topología sin puntos no busca
reemplazar a la topología clásica,
sino ampliar nuestra manera de
comprender sus construcciones
y propiedades.

\end{block}

}

\end{frame}
