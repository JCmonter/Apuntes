    \fill (0,0) circle (2pt) node[below] {$x$};
    \fill (1.2,0.8) circle (2pt) node[right] {$y$};

\[
\mathcal O(X)
=
\left\{
\varnothing,\,
\{z\},\,
\{y,z\},\,
X
\right\}.
\]

\centering

\begin{tikzpicture}[
    node distance=0.65cm,
    every node/.style={inner sep=1pt}
]
\node (X) {$X$};
\node[below=of X] (yz) {$\{y,z\}$};
\node[below=of yz] (z) {$\{z\}$};
\node[below=of z] (empty) {$\varnothing$};

\draw (X) -- (yz);
\draw (yz) -- (z);
\draw (z) -- (empty);
\end{tikzpicture}

\end{column}

\end{columns}
