\begin{frame}{Derivadas y núcleos espacialmente inducidos}

Sea \(S\) un espacio topológico y \(E\subseteq S\).

\vspace{0.1cm}

\begin{columns}[T,onlytextwidth]

    %--------------------------------
    \column{0.48\textwidth}

    \begin{block}{Derivada}
    \[
        \partial_E\colon\mathcal{C}S\longrightarrow\mathcal{C}S
    \]
    es una \textbf{derivada} si es:
    \begin{itemize}
        \item monótono,
        \item desinflacionario,
        \item preserva uniones.
    \end{itemize}

    Si además
    \[
        \partial_E^2=\partial_E,
    \]
    es una \textbf{derivada idempotente}.
    \end{block}

    %--------------------------------
    \column{0.48\textwidth}

    \begin{block}{Complemento dual}
    \[
        [E]\colon\mathcal{O}S\longrightarrow\mathcal{O}S
    \]
    es:
    \begin{itemize}
        \item monótono,
        \item inflacionario,
        \item preserva intersecciones.
    \end{itemize}

    Si \(\partial_E\) es idempotente, entonces
    \[
        [E]^2=[E],
    \]
    y \([E]\) es un \textbf{núcleo espacialmente inducido}.
    \end{block}

\end{columns}

\vspace{0.1cm}

\begin{center}
    \small
    \[
    \boxed{
    \begin{array}{ccccc}
    \text{Derivada}
        & \longleftrightarrow &
    \text{Prenúcleo}
    \\[1mm]
    \text{Derivada idempotente}
        & \longleftrightarrow &
    \text{Núcleo espacialmente inducido}
    \end{array}}
    \]
    \vspace{-0.2cm}

    {\scriptsize La correspondencia está dada mediante complemento.}
\end{center}

\end{frame}
